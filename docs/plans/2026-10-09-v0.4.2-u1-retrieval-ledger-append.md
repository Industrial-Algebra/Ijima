### `ijima-core/src/retrieval.rs` — FULL FINAL CONTENT

```rust
//! The retrieval ledger (v0.4.2): an append-only record of what the
//! read paths actually serve. Every wake-up, search, and browse
//! response appends one [`RetrievalEvent`] per served memory, so
//! admission, starvation, and per-memory frequency are answerable
//! after the fact. Events carry ids and metadata only — never memory
//! content and never query text.

use serde::{Deserialize, Serialize};

/// Which read path served the memory, and (for wake-up) from which
/// stratum. Serialized PascalCase like the crate's other enums.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum RetrievalKind {
    /// Wake-up recency stratum (admitted first; the starvation fix).
    WakeupRecent,
    /// Wake-up lexicographic fill (importance-order after the recency
    /// stratum).
    WakeupLexicographic,
    /// The universal doctrine wall in wake-up.
    WakeupDoctrine,
    /// Semantic search (`POST /memories/search`).
    Search,
    /// Browse/list (`GET /memories`).
    Browse,
}

/// One served memory. `rank` is the 0-based position in the response
/// list it was served in; `similarity` is set for `Search` events only
/// (cosine similarity, 0.0–1.0) and absent otherwise.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct RetrievalEvent {
    /// Groups all events from one request.
    pub retrieval_id: String,
    /// The read path (and stratum, for wake-up).
    pub kind: RetrievalKind,
    /// The wall the memory was served from. For `scope=visible`
    /// search this is the contributing wall each hit merged from.
    pub namespace: String,
    /// The served memory's id (join key back to the palace).
    pub memory_id: String,
    /// The authenticated principal that triggered the retrieval.
    pub principal: String,
    /// 0-based position in the served list.
    pub rank: u32,
    /// Cosine similarity; present on `Search` events only. Skipped
    /// when `None` (a serialized null here would poison reads — the
    /// v0.4.1 lesson).
    #[serde(skip_serializing_if = "Option::is_none")]
    pub similarity: Option<f32>,
    /// Unix seconds.
    pub ts_unix: u64,
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn event_round_trips_pascal_case_kind_and_skipped_similarity() {
        let e = RetrievalEvent {
            retrieval_id: "r1".into(),
            kind: RetrievalKind::WakeupRecent,
            namespace: "ns_x".into(),
            memory_id: "m1".into(),
            principal: "elliott".into(),
            rank: 0,
            similarity: None,
            ts_unix: 100,
        };
        let j = serde_json::to_string(&e).expect("serialize");
        assert!(!j.contains("similarity"), "None similarity is skipped: {j}");
        assert!(j.contains("\"WakeupRecent\""), "PascalCase kind: {j}");
        let back: RetrievalEvent = serde_json::from_str(&j).expect("deserialize");
        assert_eq!(back, e);
    }

    #[test]
    fn event_round_trips_with_similarity() {
        let e = RetrievalEvent {
            retrieval_id: "r2".into(),
            kind: RetrievalKind::Search,
            namespace: "ns_x".into(),
            memory_id: "m2".into(),
            principal: "elliott".into(),
            rank: 3,
            similarity: Some(0.75),
            ts_unix: 100,
        };
        let j = serde_json::to_string(&e).expect("serialize");
        assert!(j.contains("\"similarity\":0.75"), "{j}");
        let back: RetrievalEvent = serde_json::from_str(&j).expect("deserialize");
        assert_eq!(back, e);
    }
}
```

### `ijima-server/src/api.rs`

1. Helper near the top of the handler section (after the constants):

```rust
/// Unique-per-process retrieval ids: nanos + a monotonic counter —
/// no new dependency, collision-free within and across restarts
/// (nanos differ across restarts by wall clock).
fn new_retrieval_id() -> String {
    use std::sync::atomic::{AtomicU64, Ordering};
    static N: AtomicU64 = AtomicU64::new(0);
    let seq = N.fetch_add(1, Ordering::SeqCst);
    let nanos = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_nanos())
        .unwrap_or(0);
    format!("rl_{nanos:x}_{seq:x}")
}

fn retrieval_ts() -> u64 {
    std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0)
}
```

2. **Wake-up hook.** In `async fn wakeup`, replace the composition
block (from `let mut seen = ...` through the closing
`personal_essentials.truncate(WAKEUP_PERSONAL_LIMIT);`) with:

```rust
    let mut seen = std::collections::HashSet::new();
    let mut personal_essentials = Vec::with_capacity(WAKEUP_PERSONAL_LIMIT);
    // Stratum tags ride the composition (v0.4.2 ledger): the recency
    // stratum is admitted first, the lexicographic fill second; the
    // tag is what the ledger records per served memory.
    let mut stratum_of: std::collections::HashMap<String, ijima_core::RetrievalKind> =
        std::collections::HashMap::new();
    for m in recent.map_err(internal)? {
        if seen.insert(m.id.0.clone()) {
            stratum_of.insert(m.id.0.clone(), ijima_core::RetrievalKind::WakeupRecent);
            personal_essentials.push(m);
        }
    }
    for m in top.map_err(internal)? {
        if seen.insert(m.id.0.clone()) {
            stratum_of.insert(m.id.0.clone(), ijima_core::RetrievalKind::WakeupLexicographic);
            personal_essentials.push(m);
        }
    }
    personal_essentials.truncate(WAKEUP_PERSONAL_LIMIT);

    let doctrine = doctrine.map_err(internal)?;
    // The ledger (v0.4.2): fire-and-logged — an append failure warns
    // and the wake-up still succeeds. Observability must not become a
    // new failure mode for reads.
    let rid = new_retrieval_id();
    let ts = retrieval_ts();
    let mut ledger: Vec<ijima_core::RetrievalEvent> = Vec::with_capacity(
        personal_essentials.len() + doctrine.len(),
    );
    for (rank, m) in personal_essentials.iter().enumerate() {
        ledger.push(ijima_core::RetrievalEvent {
            retrieval_id: rid.clone(),
            kind: *stratum_of.get(m.id.0.as_str()).unwrap_or(&ijima_core::RetrievalKind::WakeupLexicographic),
            namespace: essentials_ns.as_str().to_string(),
            memory_id: m.id.0.clone(),
            principal: principal.0.principal.as_str().to_string(),
            rank: rank as u32,
            similarity: None,
            ts_unix: ts,
        });
    }
    for (rank, m) in doctrine.iter().enumerate() {
        ledger.push(ijima_core::RetrievalEvent {
            retrieval_id: rid.clone(),
            kind: ijima_core::RetrievalKind::WakeupDoctrine,
            namespace: doctrine_ns.as_str().to_string(),
            memory_id: m.id.0.clone(),
            principal: principal.0.principal.as_str().to_string(),
            rank: rank as u32,
            similarity: None,
            ts_unix: ts,
        });
    }
    if let Err(e) = store.append_retrieval_events(&ledger).await {
        tracing::warn!(error = %e, "retrieval ledger append failed (wakeup)");
    }

    Ok(Json(WakeupResponse {
        identity: serde_json::json!({ "principal": principal.0.principal.as_str() }),
        personal_essentials,
        doctrine,
    }))
```

Note: `doctrine` is bound once (the old code consumed it in the
response; the hook needs it first, then moves it into the response).
